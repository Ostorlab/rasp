cmake_minimum_required(VERSION 4.4)
project(entry_host LANGUAGES C CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(runtime "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_subdirectory("${runtime}/entrypoint_validation" entry-source)
add_executable(entry_c ../consumer.c ../c_host.c)
target_include_directories(entry_c PRIVATE "${RASP_POLICY_INCLUDE}" "${CMAKE_CURRENT_LIST_DIR}/..")
target_link_libraries(entry_c PRIVATE rasp_entrypoint_validation "${RASP_LIBRARY}")
target_compile_options(entry_c PRIVATE -Wall -Wextra -Werror -Wpedantic -Wconversion -Wframe-larger-than=4096)
set_target_properties(entry_c PROPERTIES C_STANDARD 11 C_VISIBILITY_PRESET hidden)
add_executable(entry_cpp ../consumer_cpp.cpp ../c_host.c)
target_include_directories(entry_cpp PRIVATE "${RASP_POLICY_INCLUDE}" "${CMAKE_CURRENT_LIST_DIR}/..")
target_link_libraries(entry_cpp PRIVATE rasp_entrypoint_validation "${RASP_LIBRARY}")
target_compile_options(entry_cpp PRIVATE -Wall -Wextra -Werror -Wpedantic -Wconversion -Wframe-larger-than=4096
  "$<$<COMPILE_LANGUAGE:CXX>:-nostdinc++>")
set_target_properties(entry_cpp PROPERTIES CXX_STANDARD 23 CXX_VISIBILITY_PRESET hidden C_VISIBILITY_PRESET hidden)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  add_library(entry_url OBJECT "${runtime}/entrypoint_validation/ios/url_bytes.c")
  target_link_libraries(entry_url PRIVATE rasp_entrypoint_validation)
  target_compile_options(entry_url PRIVATE -Wall -Wextra -Werror -Wpedantic -Wconversion -Wframe-larger-than=4096)
  set_target_properties(entry_url PROPERTIES C_STANDARD 11 C_VISIBILITY_PRESET hidden)
endif()
