cmake_minimum_required(VERSION 4.4)
project(HostContextConsumers LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(runtime "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
if(NOT EXISTS "${RASP_LIBRARY}" OR NOT IS_DIRECTORY "${RASP_POLICY_INCLUDE}")
  message(FATAL_ERROR "Supply the actual runtime library and matching public header directory")
endif()
if(ANDROID)
  set(host_clock "${runtime}/host_context/android/host_clock.c")
elseif(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set(host_clock "${runtime}/host_context/ios/host_clock.c")
else()
  message(FATAL_ERROR "Actual Android NDK or Apple iOS toolchain required")
endif()
add_library(host_context STATIC "${runtime}/host_context/src/host_context.c"
  "${runtime}/host_context/src/clock_math.c" "${host_clock}")
target_include_directories(host_context PUBLIC "${runtime}/host_context/include" "${RASP_POLICY_INCLUDE}")
set_target_properties(host_context PROPERTIES C_VISIBILITY_PRESET hidden)
target_compile_options(host_context PRIVATE -Wall -Wextra -Werror -Wpedantic -Wconversion -Wshadow -Wframe-larger-than=4096)
foreach(language c cpp)
  if(language STREQUAL "c")
    set(consumer "${runtime}/tests/host_context/c_host.c")
  else()
    set(consumer "${runtime}/tests/host_context/cpp_host.cpp")
  endif()
  add_executable(host_${language} "${consumer}")
  target_compile_options(host_${language} PRIVATE -Wall -Wextra -Werror -Wpedantic -Wconversion -Wshadow)
  if(language STREQUAL "cpp")
    target_compile_options(host_${language} PRIVATE -nostdinc++)
  endif()
  target_link_libraries(host_${language} PRIVATE host_context "${RASP_LIBRARY}")
  if(ANDROID)
    target_link_options(host_${language} PRIVATE -Wl,--allow-shlib-undefined)
  endif()
endforeach()
